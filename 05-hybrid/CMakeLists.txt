cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(UFAZ_GameOfLife LANGUAGES C)
else()
  enable_language(C)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_executable(life_serial life.c)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(life_serial PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/life_hybrid.c")
  if((NOT DEFINED PW_ENABLE_MPI OR PW_ENABLE_MPI) AND
     (NOT DEFINED PW_ENABLE_OPENMP OR PW_ENABLE_OPENMP))
    find_package(MPI REQUIRED COMPONENTS C)
    find_package(OpenMP REQUIRED COMPONENTS C)
    add_executable(life_hybrid life_hybrid.c)
    target_link_libraries(life_hybrid PRIVATE MPI::MPI_C OpenMP::OpenMP_C)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
      target_compile_options(life_hybrid PRIVATE -Wall -Wextra -Wpedantic)
    endif()
  endif()
endif()
